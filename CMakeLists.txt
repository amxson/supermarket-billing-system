cmake_minimum_required(VERSION 3.16)
project(supermarket_billing LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(inventory product.cpp category.cpp files.cpp)
target_include_directories(inventory PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(supermarket main.cpp cart.cpp)
target_link_libraries(supermarket PRIVATE inventory)
enable_testing()
add_executable(persistence_test tests/persistence.cpp)
target_link_libraries(persistence_test PRIVATE inventory)
add_test(NAME persistence COMMAND persistence_test)
